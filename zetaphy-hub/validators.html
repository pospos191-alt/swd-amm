C=$OCI_TENANCY
AD=$(oci iam availability-domain list -c $C --query 'data[0].name' --raw-output)
SUB=$(oci network subnet list -c $C --query 'data[?"prohibit-public-ip-on-vnic"==`false`] | [0].id' --raw-output)
img() { oci compute image list -c $C --operating-system "Canonical Ubuntu" --operating-system-version "24.04" --shape $1 --sort-by TIMECREATED --query 'data[0].id' --raw-output; }
IA=$(img VM.Standard.A1.Flex); IM=$(img VM.Standard.E2.1.Micro)
[ -f ~/.ssh/id_rsa.pub ] || ssh-keygen -t rsa -b 4096 -N '' -q -f ~/.ssh/id_rsa
for v in AD SUB IA IM; do echo "$v=${!v:0:40}"; done
try() { oci compute instance launch --no-retry -c $C --availability-domain $AD --subnet-id $SUB --assign-public-ip true \
  --display-name zetaphy-validator --ssh-authorized-keys-file ~/.ssh/id_rsa.pub "$@" --query 'data.id' --raw-output 2>~/launch.err; }
n=0; while [ -n "$SUB" ]; do n=$((n+1))
  if [ $((n % 2)) = 1 ]; then ID=$(try --shape VM.Standard.A1.Flex --shape-config '{"ocpus":1,"memoryInGBs":6}' --image-id $IA) || ID=
  else ID=$(try --shape VM.Standard.E2.1.Micro --image-id $IM) || ID=; fi
  [ -n "$ID" ] && { echo "GOT $ID"; break; }
  echo "$(date +%H:%M) round $n: $(grep -o '"message": "[^"]*' ~/launch.err | head -1 | cut -c13-90)"; sleep 300
done
[ -n "$ID" ] && { sleep 90; echo "ssh ubuntu@$(oci compute instance list-vnics --instance-id $ID --query 'data[0]."public-ip"' --raw-output)"; }
